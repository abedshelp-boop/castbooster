```python
class _FlushingRotatingFileHandler(RotatingFileHandler):
    """Rotating handler that fsyncs after every record.

    Trade ~5% perf for not losing the last ~64 KB on crash. Pillar 3.5
    crash diagnostics: silent deaths must leave evidence behind.

    def emit(self, record):
        super().emit(record)
        try:
            self.flush()
            if self.stream is not None:
                os.fsync(self.stream.fileno())
        except Exception:
            pass  # logging must never raise from inside emit
```

```python
def _asyncio_exception_handler(loop, context):
    """Log unhandled asyncio coroutine exceptions before swallowing.

    Pillar 3.5 thread 1: asyncio's default handler logs to its own
    'asyncio' logger which we DON'T capture in the rotating file
    handler. Route through 'castbooster' so it lands in castbooster.log.

    exc = context.get("exception")
    msg = context.get("message", "unhandled asyncio context")
    if exc is not None:

            "FATAL unhandled asyncio exception: %s", msg,

        log.error("FATAL unhandled asyncio context: %s | full=%r", msg, context)
```

```python
def _build_filter_chain(
    enable_smooth: bool,
    caps,
    video,
) -> tuple[FilterChain, Optional[str]]:
    """Spec §4.2 decision tree.

    Returns (chain, info_message_or_None). info_message surfaces to the
    popup via the cast/set_filter_chain response so the user sees WHY
    smoothness was skipped.

    Pillar 3.5: probe-failure case now reports _PROBE_FAILED_INFO instead
    of silently demoting (was silent in P3.4).
    """
    # User asked for smooth + has Pro + Chromecast supports video, but ffprobe
    # couldn't tell us the source FPS. Surface that explicitly.
    if (
        video is None
        and enable_smooth
        and license.is_pro()
        and not caps.audio_only
    ):
        return FilterChain([NoopFilter()]), _PROBE_FAILED_INFO
    # Catch-all silent fallback (audio-only / toggle off / not pro / no video).
    if caps.audio_only or not enable_smooth or not license.is_pro() or video is None:
        return FilterChain([NoopFilter()]), None
    target_fps = caps.max_fps
    if video.fps is None:
        return FilterChain([NoopFilter()]), _VFR_INFO
    if video.fps >= target_fps:
        return FilterChain([NoopFilter()]), None
    rife = RIFEFilter(
        source_fps=video.fps, target_fps=target_fps,
        width=video.width, height=video.height,
    )
    return FilterChain([rife]), None
```

```python
        return FilterChain([NoopFilter()]), _PROBE_FAILED_INFO

        return FilterChain([NoopFilter()]), None

        return FilterChain([NoopFilter()]), None
    if not license.is_pro():
        log.info("_build_filter_chain: branch=not_pro")
        return FilterChain([NoopFilter()]), None
    if video is None:
        log.info("_build_filter_chain: branch=no_video (smooth=%s)", enable_smooth)
        return FilterChain([NoopFilter()]), None
```

```markdown
# Server-side migration — deferred until post-launch

During the 2026-05-20 brainstorming around Pillar 3.5, Abed considered
pivoting Castbooster from "local PC does the RIFE work" to "our cloud
servers do the work." After laying out the cost / complexity / legal
implications, he chose to **defer** the migration and continue building
Pillars 3/4 on the local PC architecture. This note captures the
deferral so a future session doesn't re-derive the analysis from scratch.

## Decision

Continue local-PC architecture for Pillars 3, 4, 5, and 6. Ship v1 as
the Windows desktop product. Revisit server-side AFTER seeing real
user demand and local-quality results.

## Trigger conditions to revisit

- Mac / iOS / Android / Chromebook demand from real customers (>10% of
  post-launch feature requests).
- Evidence that >10% of Windows users fall through to passthrough on a
  post-launch hardware survey (i.e. local RIFE isn't reaching enough
  of the install base).
- Local-quality plateau — Pillar 3 + 4 ship but RIFE on consumer
  hardware proves insufficient for the target visual quality.

## Architecture sketch (if/when we revisit)

- **Cloud GPU pool**: Modal / RunPod / Replicate / fly.io GPU instances.
  Per-user-hour cost estimate $0.10–$0.50 GPU + $0.15–$0.60 egress =
  **$0.25–$1.10 per user per viewing hour**.
- **Stream proxy server**: Go or Rust on Linux, handling session-locked
  CDN fetches with residential proxies ($5–$15/GB additional cost).
- **Account + billing**: Polar.sh (already planned for Pillar 6), but
  pricing must shift to subscription ($10–$20/month) to cover infra.
- **Latency budget**: 1–3 seconds of buffering on our server to give RIFE
  time to process. Acceptable for VOD; unsuitable for live streams.
- **Legal posture**: shifts from "VLC-like, content never touches our
  servers" to "active middleman transmitting third-party content."
  DMCA safe-harbor protections get fuzzy; review with counsel before
  shipping.
- **Engineering effort**: ~6–12 months on top of the local product.

## Why not now

- 95% of typical Intel / AMD / NVIDIA Windows laptops (the v1 target
  market) already have HW encoders + Vulkan iGPUs that can run RIFE
  locally with reasonable load. The local product is sufficient for
  the audience we're targeting.
- Server-side burns infrastructure cash from day 1, before we know
  whether the product actually has demand.
- The 6–12 month engineering shift would push public launch into 2027.
  Better to ship the local product, learn what users want, then decide
  whether server-side justifies the investment.

## What replaces it for now

- Pillar 3.5 thread 3 (h264_mf encoder support) covers ARM Windows
  users — the ~3% slice that was the most painful local-hardware case.
- Pillar 5 (auto-detect + watchdog) will handle graceful "your hardware
  can't do this" messaging for the remaining edge cases.

## Sources

- 2026-05-20 brainstorming session (this Claude Code conversation).
- Cost estimates: AWS / GCP GPU instance public pricing + ChatGPT/Modal
  inference cost benchmarks (rough order of magnitude).
- Engineering estimate: parallels to similar pivots (e.g. Stadia,
  GeForce-NOW go-to-market timelines).
```
